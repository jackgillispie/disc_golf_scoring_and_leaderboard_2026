        <script>
            // Automatically load data on page load
            window.onload = function () {
                google.script.run
                    .withSuccessHandler(drawTable)
                    .getLeaderboardData();
                // google.script.run.withFailureHandler().onFailure();
            };

            function drawTable(data) {
                if (!data || data.length === 0) {
                    document.getElementById("board").innerHTML =
                        "No data found.";
                    return;
                }

                let html = `
                <div class="row">
                <span class="rank">Rank</span>
                <span class="name">Names</span>
                <span class="score">Score</span>
                </div>`;

                data.forEach((row, index) => {
                    html += `
                <div class="row">
                <span class="rank">#${row[0]}</span>
                <span class="name">${row[1]}</span>
                <span class="score">${row[2]}</span>
                </div>`;
                });
                document.getElementById("board").innerHTML = html;

                // color 1st, second, and withFailureHandler
                top_three = document.getElementsByClassName("row");
                top_three[1].style.background = "#FFD700";
                top_three[2].style.background = "#FF1493";
                top_three[3].style.background = "#0077BE";
            }
